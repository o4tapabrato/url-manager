const handleCreateLink = async (e) => {
        e.preventDefault();
        if (!originalUrl) return;

        try {
            const response = await fetch('/api/links', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    originalUrl,
                    customSlug: customSlug || undefined,
                }),
            });

            const data = await response.json();

            if (!response.ok) {
                throw new Error(data.error || 'Failed to create short link');
            }

            // Construct the formatted object for your state table
            const newLink = {
                id: data.link.id,
                title: data.link.shortCode,
                originalUrl: data.link.originalUrl,
                shortCode: data.link.shortCode,
                shortUrl: `${baseUrl}/${data.link.shortCode}`,
                clicks: data.link.clicks,
                createdAt: data.link.createdAt.split('T')[0],
            };

            setLinks([newLink, ...links]);
            setOriginalUrl('');
            setCustomSlug('');
        } catch (error) {
            alert(error.message);
        }
    };
